from pyecharts.charts import *
from pyecharts import options as opts
from pyecharts.faker import Faker

color_js = """new echarts.graphic.LinearGradient(0, 1, 0, 0,
    [{offset: 0, color: 'red'}, {offset: 1, color: 'pink'}], false)""" 

sort_info = df.sort_values(by='销量', ascending=True)  # 对销量进行升序排序
def bar_reverse_axis():
    bar = Bar(init_opts=opts.InitOpts(theme='light',
                                      width='1000px',
                                      height='600px'))
    bar.add_xaxis(list(sort_info['名称'])[-20:])
    # 修改颜色为蓝色，直接指定颜色值
    bar.add_yaxis('', sort_info['销量'].tolist()[-20:],itemstyle_opts=opts.ItemStyleOpts(color=JsCode(color_js)))  # 红色
  
    bar.reversal_axis()  	# x,y轴翻转，方便名称填充
    bar.set_global_opts(
    title_opts=opts.TitleOpts(title='热门景点销量数据'),
    yaxis_opts=opts.AxisOpts(name='景点名称'),
    xaxis_opts=opts.AxisOpts(name='销量'),
    )
    bar.set_series_opts(label_opts=opts.LabelOpts(position="right"))  # left  right center
    return bar

chart = bar_reverse_axis()
chart.render_notebook()

注

dict.get(key, default)

去字典 dictcode 里找键 i：

如果 i 在 dictcode 中存在 → 返回 dictcode[i]

如果 i 不存在 → 返回 i 本身（而不是报错或 None）

```
dictcode = {1: 10, 2: 20}

print(dictcode.get(1, 1))  # 存在 → 10
print(dictcode.get(3, 3))  # 不存在 → 返回 3
print(dictcode.get(5, 5))  # 不存在 → 返回 5
```

### 景点简介词云

"D:\博客\python\数据分析\实战\辅助图片等\16图片\stopwords.txt"作为自定义词典

```
 # 要把所有的行数合并成一个字符串
contents = "".join( i for i in df['简介'].values.tolist())
contents_list = jieba.cut(contents)   # 通过jieba将字符串进行分词

ac = Counter(contents_list)

stopwords = []

with open(r'D:\博客\python\数据分析\实战\辅助图片等\16图片\stopwords.txt', "r",encoding='utf-8') as f:  # 打开文件
    data = f.read()  # 读取文件
    stopwords = data.split('\n')

for i in stopwords:
    del ac[i]
 
# 数据展示
w1 = (
    WordCloud()
    .add("", 
         ac.most_common(100), 			# 展示前多少个数据
         word_size_range=[5, 100], 
         textstyle_opts=opts.TextStyleOpts(font_family="cursive"),
         shape='star')
    .set_global_opts(title_opts=opts.TitleOpts(title="景点简介词云"))
)
w1.render_notebook()
```
